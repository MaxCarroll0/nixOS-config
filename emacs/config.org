(defun my/reload-emacs-config ()
  "Tangle and load the complete working-tree Emacs configuration."
  (interactive)
  (require 'org)
  (require 'ob-tangle)
  (let* ((source "/home/max/.config/nix/emacs/config.org")
         (source-buffer (get-file-buffer source)))
    (unless (file-readable-p source)
      (user-error "Emacs configuration is not readable: %s" source))
    (let* ((source-text
            (if (buffer-live-p source-buffer)
                (with-current-buffer source-buffer
                  (save-restriction
                    (widen)
                    (buffer-substring-no-properties
                     (point-min) (point-max))))
              (with-temp-buffer
                (insert-file-contents source)
                (buffer-string))))
           (source-modified
            (and (buffer-live-p source-buffer)
                 (buffer-modified-p source-buffer)))
           (source-tick
            (and (buffer-live-p source-buffer)
                 (with-current-buffer source-buffer
                   (buffer-chars-modified-tick))))
           (directory (let ((cache (locate-user-emacs-file "config-reload/")))
                        (make-directory cache t)
                        cache))
           (copy (expand-file-name "config.org" directory))
           (elisp (expand-file-name "config.el" directory)))
      (unwind-protect
          (progn
            (when (buffer-live-p source-buffer)
              (with-current-buffer source-buffer
                (set-buffer-modified-p nil)))
            (with-temp-buffer
              (insert source-text)
              (setq buffer-file-name copy
                    default-directory directory)
              (org-mode)
              (org-babel-tangle)
              (set-buffer-modified-p nil)
              (setq buffer-file-name nil))
            (let ((byte-compile-verbose nil)
                  (byte-compile-warnings nil))
              (byte-compile-file elisp))
            (let ((default-directory (file-name-directory source))
                  (envrc--lighter nil)
                  (inhibit-message t))
              (load (if (file-readable-p (concat elisp "c"))
                        (concat elisp "c")
                      elisp)
                    nil nil t))
            (message "Reloaded complete Emacs configuration"))
        (when (buffer-live-p source-buffer)
          (with-current-buffer source-buffer
            (set-buffer-modified-p
             (or source-modified
                 (/= source-tick (buffer-chars-modified-tick))))))
        ))))

Direnv uses the async =envrc= PR referenced by Tyler Langlois' [[https://blog.tjll.net/asynchronous-emacs-direnv-devshells/][Asynchronous Emacs Direnv Devshells]], with Eglot deferred until the buffer environment has been applied.
#+begin_src emacs-lisp

(use-package envrc
  :config
  (setq envrc-async-processing t)

  (defvar-local max/envrc-deferred-eglot nil
    "Deferred `eglot-ensure' call waiting for envrc.")

  (defun max/envrc-loading-p ()
    "Return non-nil when current buffer's envrc is still loading."
    (and (bound-and-true-p envrc-mode)
         (fboundp 'envrc--find-env-dir)
         (envrc--find-env-dir)
         (not (memq envrc--status '(on error denied)))))

  (defun max/envrc-defer-eglot-ensure (orig &rest args)
    "Defer ORIG until envrc has applied the buffer environment."
    (if (max/envrc-loading-p)
        (setq max/envrc-deferred-eglot
              (lambda () (apply orig args)))
      (apply orig args)))

  (defun max/envrc-run-deferred-eglot (buf &optional _result)
    "Run BUF's deferred Eglot startup after envrc finishes."
    (with-current-buffer buf
      (when max/envrc-deferred-eglot
        (let ((fn max/envrc-deferred-eglot))
          (setq max/envrc-deferred-eglot nil)
          (funcall fn)))))

  (with-eval-after-load 'eglot
    (advice-add 'eglot-ensure :around #'max/envrc-defer-eglot-ensure))
  (advice-add 'envrc--apply :after #'max/envrc-run-deferred-eglot)
  (envrc-global-mode))

* Terminal
** Eat

Pure-elisp terminal, with =eat-eshell-mode= upgrading eshell visual commands. Meow normal ↔ eat-emacs-mode, meow insert ↔ eat-semi-char-mode. =<escape>= in semi-char must be rebound because eat otherwise forwards it to the terminal. =C-l= is resent on window-size change because fullscreen TUIs cache cells and leave restored rows blank.

#+begin_src emacs-lisp

(use-package eat
  :hook ((eshell-load . eat-eshell-mode)
         (eshell-load . eat-eshell-visual-command-mode)
         (eat-mode . my/disable-line-numbers)
         (eshell-mode . my/disable-line-numbers))
  :config
  (defun my/eat-meow-mode-sync ()
    "Sync eat input mode with meow state."
    (when (derived-mode-p 'eat-mode)
      (if (meow-normal-mode-p)
          (eat-emacs-mode)
        (eat-semi-char-mode))))
  (add-hook 'meow-normal-mode-hook #'my/eat-meow-mode-sync)
  (add-hook 'meow-insert-mode-hook #'my/eat-meow-mode-sync)
  (add-hook 'eat-mode-hook #'my/eat-meow-mode-sync)

  (define-key eat-semi-char-mode-map (kbd "<escape>") #'meow-insert-exit)
  (define-key eat-eshell-semi-char-mode-map (kbd "<escape>") #'meow-insert-exit)

  (defun my/eat-redraw-on-resize (window)
    "Force a full repaint of the eat program in WINDOW."
    (with-selected-window window
      (when (and (derived-mode-p 'eat-mode) eat-terminal)
        (eat-term-send-string eat-terminal "\C-l"))))
  (add-hook 'window-size-change-functions
            (lambda (frame)
              (walk-windows #'my/eat-redraw-on-resize nil frame))))

** GPTel
Interactive LLM chat uses ChatGPT OAuth with Emacs tools available.
Global and project instructions are added as context before requests.
Every request tells the model where its response will land (buffer at
point, chat buffer, echo area, kill-ring, minibuffer).

#+begin_src emacs-lisp

  (use-package gptel
    :bind (("C-c g" . gptel-menu)
           ("C-c G" . gptel-toggle-auto-agents-context))
    :config
    (require 'cl-lib)
    (require 'seq)
    (require 'gptel-context)

    (setq gptel-default-mode 'org-mode
          gptel-backend (gptel-make-openai-oauth "ChatGPT")
          gptel-model 'gpt-5.4-mini)

    (defvar gptel-auto-agents-context t)

    (defun gptel-toggle-auto-agents-context ()
      "Toggle automatic AGENTS.md context for gptel."
      (interactive)
      (setq gptel-auto-agents-context
            (not gptel-auto-agents-context))
      (unless gptel-auto-agents-context
        (gptel-remove-agents-context (current-buffer)))
      (message "GPTel AGENTS.md context %s"
               (if gptel-auto-agents-context "enabled" "disabled")))

    (defun gptel-request-buffer (args)
      "Return the request buffer named by ARGS."
      (or (plist-get (cdr args) :buffer)
          (current-buffer)))

    (defun gptel-agents-files (buffer)
      "Return global and project AGENTS.md files for BUFFER."
      (let* ((global (expand-file-name "~/.codex/AGENTS.md"))
             (default-directory
              (buffer-local-value 'default-directory buffer))
             (local-dir (locate-dominating-file default-directory "AGENTS.md"))
             (local (and local-dir (expand-file-name "AGENTS.md" local-dir))))
        (delete-dups
         (mapcar #'file-truename
                 (seq-filter #'file-readable-p
                             (delq nil (list global local)))))))

    (defun gptel-remove-agents-context (buffer)
      "Remove AGENTS.md files from gptel context for BUFFER."
      (with-current-buffer buffer
        (dolist (file (gptel-agents-files buffer))
          (setq gptel-context
                (cl-remove-if
                 (lambda (item)
                   (equal item (list file)))
                 gptel-context)))))

    (defun gptel-add-agents-context (orig &rest args)
      "Add AGENTS.md files to gptel context before calling ORIG with ARGS."
      (let ((buffer (gptel-request-buffer args)))
        (when gptel-auto-agents-context
          (with-current-buffer buffer
            (dolist (file (gptel-agents-files buffer))
              (cl-pushnew (list file) gptel-context :test #'equal)))))
      (apply orig args))

    (advice-add #'gptel-request :around #'gptel-add-agents-context)

    (defvar gptel-menu-redirect nil)

    (defun gptel-note-menu-redirect (orig args)
      "Bind the menu output redirect while ORIG runs on ARGS."
      (let ((gptel-menu-redirect
             (cond ((member "e" args) "shown in the echo area")
                   ((member "k" args) "copied to the kill-ring"))))
        (funcall orig args)))

    (defun gptel-buffer-description (buffer)
      "Describe BUFFER's file, directory, and active modes."
      (with-current-buffer buffer
        (let ((minor-modes
               (seq-filter
                (lambda (mode)
                  (and (boundp mode) (symbol-value mode)))
                (mapcar #'car minor-mode-alist))))
          (format "%s; file %s; default-directory %s; major mode %s; active minor modes %s"
                  (buffer-name)
                  (or buffer-file-name "none")
                  default-directory
                  major-mode
                  (if minor-modes
                      (mapconcat #'symbol-name minor-modes ", ")
                    "none")))))

    (defun gptel-destination-buffer (info)
      "Return the response destination buffer for request INFO."
      (let ((position (plist-get info :position)))
        (or (and (markerp position)
                 (marker-buffer position))
            (plist-get info :buffer))))

    (defun gptel-destination-description (info)
      "Describe where the response for request INFO will end up."
      (or gptel-menu-redirect
          (when-let* ((callback (plist-get info :callback))
                      ((not (memq callback '(gptel--insert-response
                                             gptel-curl--stream-insert-response)))))
            "passed to an Emacs Lisp callback, not inserted into a buffer")
          (when-let* ((buffer (gptel-destination-buffer info)))
            (with-current-buffer buffer
              (concat
               (cond ((minibufferp) "inserted into the minibuffer")
                     (gptel-mode
                      (format "appended to the dedicated gptel chat buffer %s"
                              (buffer-name)))
                     (buffer-file-name
                      (format "inserted at point in buffer %s (%s, visiting %s)"
                              (buffer-name) major-mode buffer-file-name))
                     (t (format "inserted at point in buffer %s (%s)"
                                (buffer-name) major-mode)))
               (and (plist-get info :in-place)
                    ", replacing the prompt text"))))))

    (defun gptel-add-destination-context (fsm)
      "Tell the model where the response for FSM will be displayed."
      (let* ((info (gptel-fsm-info fsm))
             (request-buffer (plist-get info :buffer))
             (destination-buffer (gptel-destination-buffer info)))
        (gptel-context--wrap-in-buffer
         (string-join
          (delq nil
                (list
                 (format "Response destination: your reply will be %s."
                         (gptel-destination-description info))
                 (when request-buffer
                   (format "Request buffer: %s."
                           (gptel-buffer-description request-buffer)))
                 (when destination-buffer
                   (format "Destination buffer: %s."
                           (gptel-buffer-description destination-buffer)))))
          "\n")
         'system)))

    (add-hook 'gptel-prompt-transform-functions
              #'gptel-add-destination-context 90)

    (advice-add #'gptel--suffix-send :around #'gptel-note-menu-redirect))

  (use-package llm-tool-collection
    :ensure nil
    :after gptel
    :config
    (setq gptel-tools
          (mapcar (apply-partially #'apply #'gptel-make-tool)
                  (llm-tool-collection-get-all))))

(defun my/completion-symbol-p (candidate)
  "Return non-nil when CANDIDATE starts with punctuation rather than a letter."
  (and (> (length candidate) 0)
       (not (string-match-p "\\`[[:alnum:]_]" candidate))))

(defun my/completion-sort-symbols-first (candidates)
  "Sort CANDIDATES by length and alphabetically, symbols before letters."
  (let ((sorted (corfu-sort-length-alpha candidates)))
    (nconc (seq-filter #'my/completion-symbol-p sorted)
           (seq-remove #'my/completion-symbol-p sorted))))

       (> (length candidate) 0)
       (get-text-property 0 'eglot--lsp-item candidate)))

(defun my/completion-candidate-detail (candidate)

  "Return non-nil after a successful typed or untyped probe."

(defun my/type-completion-entry (candidate)
  "Return CANDIDATE's OCaml semantic completion entry."
  (when (and (stringp candidate) (> (length candidate) 0))
    (or (get-text-property 0 'my/type-search-entry candidate)
        (and my/type-completion-index
             (gethash (substring-no-properties candidate)
                      my/type-completion-index)))))

(defun my/type-normalize (type)
  "Normalize whitespace and Markdown fences in TYPE."
  (when (stringp type)
    (let ((text type))
      (when (string-search "```" text)
        (setq text (replace-regexp-in-string
                    "```[^\n]*\n?\\|```" "" text t t)))
      (string-trim (replace-regexp-in-string "[[:space:]]+" " " text t t)))))

(defun my/type-remove-candidate-prefix (type candidate)
  "Remove CANDIDATE's optional signature prefix from TYPE."
  (let* ((type (my/type-normalize type))
         (name (and candidate
                    (regexp-quote (substring-no-properties candidate))))
         (prefix
          (and name
               (format "\\`[[:space:]]*\\(?:val[[:space:]]+\\|let[[:space:]]+\\)?%s[[:space:]]*\\(?:::\\|:\\)[[:space:]]*"
                       name))))
    (if (and prefix type (string-match prefix type))
        (substring type (match-end 0))
      type)))

(defun my/type-treesit-error-p (node)
  "Return non-nil when NODE contains a parser error."
  (or (treesit-node-check node 'has-error)
      (treesit-node-check node 'missing)
      (string= (treesit-node-type node) "ERROR")))

(defun my/type-enclosed-p (text)
  "Return non-nil when TEXT is wrapped in one balanced pair of parentheses."
  (and (> (length text) 1)
       (eq (aref text 0) ?\()
       (eq (aref text (1- (length text))) ?\))
       (let ((depth 0)
             (inner t))
         (dotimes (index (length text))
           (pcase (aref text index)
             (?\( (setq depth (1+ depth)))
             (?\) (setq depth (1- depth))))
           (when (and (zerop depth) (< index (1- (length text))))
             (setq inner nil)))
         inner)))

(defun my/type-strip-parens (text)
  "Return TEXT with any fully enclosing parentheses removed."
  (if (my/type-enclosed-p text)
      (my/type-strip-parens (string-trim (substring text 1 -1)))
    text))

(defun my/type-arrow-split (text)
  "Split TEXT on its top-level function arrows."
  (let ((depth 0)
        (start 0)
        (position 0)
        (length (length text))
        parts)
    (while (< position length)
      (pcase (aref text position)
        ((or ?\( ?\[ ?{) (setq depth (1+ depth)))
        ((or ?\) ?\] ?}) (setq depth (max 0 (1- depth))))
        ((or ?- ?=)
         (when (and (zerop depth)
                    (< (1+ position) length)
                    (eq (aref text (1+ position)) ?>))
           (push (string-trim (substring text start position)) parts)
           (setq position (1+ position)
                 start (1+ position)))))
      (setq position (1+ position)))
    (push (string-trim (substring text start)) parts)
    (nreverse parts)))

(defun my/type-arrow-parts (type)
  "Split TYPE on its top-level function arrows, flattening its codomain."
  (when-let* ((text (my/type-normalize type)))
    (let* ((parts (my/type-arrow-split (my/type-strip-parens text)))
           (codomain (car (last parts)))
           (inner (and (my/type-enclosed-p codomain)
                       (my/type-arrow-split
                        (my/type-strip-parens codomain)))))
      (if (cdr inner)
          (append (butlast parts) inner)
        parts))))

(defun my/type-arrow-count (type)
  "Return TYPE's number of top-level function arrows."
  (max 0 (1- (length (my/type-arrow-parts type)))))

(defun my/type-canonical (type)

    (string-join parts " -> ")))

(defun my/type-unconstrained-p (type)

  "Return non-nil when NAME is an OCaml value operator."
  (and (stringp name)

  "Return non-nil when NAME has OCaml prefix application syntax."
  (and (my/ocaml-operator-p name)
       (or (string-prefix-p "~" name)
           (string-prefix-p "?" name)
           (and (string-prefix-p "!" name)
                (not (member name '("!=" "!==")))))))

(defun my/type-operator-template (name holes)
  "Return natural OCaml application of operator NAME to HOLES."
  (let ((arguments (length holes)))
    (cond
     ((zerop arguments) (format "( %s )" name))
     ((my/ocaml-prefix-operator-p name)
      (concat name " " (string-join holes " ")))
     ((= arguments 1) (format "( %s ) %s" name (car holes)))
     (t
      (concat (format "(%s %s %s)" (nth 0 holes) name (nth 1 holes))
              (if (> arguments 2) " " "")
              (string-join (nthcdr 2 holes) " "))))))

(defun my/type-application (name holes)
  "Return NAME applied to HOLES in its natural OCaml syntax."
  (cond
   ((my/ocaml-let-operator-p name) nil)
   ((my/ocaml-operator-p name) (my/type-operator-template name holes))
   (holes (concat "(" name " " (string-join holes " ") ")"))))

(defun my/type-entry-annotate (entry wanted rank)
  "Record ENTRY's server RANK and its application producing canonical WANTED."
  (setq entry (plist-put entry :rank rank))
  (if (memq (plist-get entry :kind) '(construct collection record))
      (progn
        (setq entry (plist-put entry :tier 0))
        (plist-put entry :arity 0))
    (let* ((name (plist-get entry :name))
           (parts (my/type-arrow-parts (plist-get entry :typ)))
           (exact (and parts wanted (my/type-suffix-arity parts wanted)))
           (server (cdr (my/type-template-holes
                         (plist-get entry :constructible))))
           (arity (or exact (and server (length server))))
           (arity (and arity
                       (max 0 (- arity
                                 (if (my/ocaml-let-operator-p name) 2 0)))))
           (holes (cond ((null arity) nil)
                        ((>= (length server) arity) (seq-take server arity))
                        (t (my/type-holes-from-parts parts arity)))))
      (setq entry (plist-put entry :arity (or arity 0)))
      (setq entry (plist-put entry :holes holes))
      (setq entry (plist-put entry :tier
                             (cond
                              ((and wanted
                                    (eq (plist-get entry :source)
                                        'type-search))
                               0)
                              ((and exact (equal exact arity)) 0)
                              ((null (plist-get entry :typ)) 1)
                              (wanted 3)
                              (t 0))))
      (setq entry (plist-put entry :template
                             (and arity (my/type-application name holes))))
      (plist-put entry :snippet
                 (and arity
                      (my/type-application
                       name (my/type-number-holes holes)))))))

               "\\_<\\(?:let\\|fun\\|function\\|match\\)\\_>" type)))))

(defun my/corfu-filter-highlight (text)
  "Return TEXT with active regexp matches highlighted."
  (let ((copy (copy-sequence text))
        (start 0))
    (while (and (< start (length copy))
                (string-match my/corfu-filter-regexp copy start))
      (let ((beginning (match-beginning 0))
            (end (match-end 0)))

          (add-face-text-property
           beginning end 'completions-common-part t copy))

(defun my/corfu-filter-refresh ()
  "Apply the active regexp directly to the Corfu popup."
  (condition-case error
      (let* ((case-fold-search completion-ignore-case)
             (current (and (>= corfu--index 0)
                           (nth corfu--index corfu--candidates)))
             (empty (string-empty-p my/corfu-filter-regexp))
             (position 0)
             (selected nil)
             (filtered nil))
        (cl-loop for candidate in my/corfu-filter-candidates
                 for value in my/corfu-filter-values
                 when (or empty (string-match-p my/corfu-filter-regexp value))
                 do (when (eq candidate current) (setq selected position))
                    (push candidate filtered)
                    (setq position (1+ position)))
        (setq filtered (nreverse filtered))
        (setq my/corfu-filter-error nil)
        (setq corfu--candidates filtered
              corfu--total (length filtered)
              corfu--scroll 0
              corfu--preselect (if filtered 0 -1)
              corfu--index (or selected corfu--preselect))
        (my/corfu-filter-popup)
        (my/corfu-filter-show-prompt))
    (invalid-regexp
     (setq my/corfu-filter-error (cadr error))
     (my/corfu-filter-show-prompt))))

** Inline preview
The best prefix-only completion tier appears inline and is accepted with =TAB=; ambiguous candidates show only their shared prefix, while a sole candidate shows its complete snippet expansion.

Cursor motion triggers the preview from an idle timer without blocking on a language-server type probe. Already-correct terms suppress it, except for bare holes.

At a bare hole, completion considers every candidate in the best tier and previews the result as a replacement for the underscore.

(defvar-local my/completion-preview-dismissed nil)
(defvar-local my/completion-preview-candidates nil)
(defvar-local my/completion-preview-hole-overlay nil)
(defvar my/completion-preview-accepting nil)
(defvar my/completion-preview-mouse-map nil)

(defun my/ocaml-incomplete-pattern-context-p ()
  "Return non-nil when nearby incomplete syntax starts a pattern."
  (save-excursion
    (let ((limit (max (point-min) (- (point) 2000))))
      (catch 'context
        (while (re-search-backward
                "\\_<\\(?:let\\|and\\|fun\\|function\\|with\\|type\\|module\\|class\\|exception\\|val\\|external\\)\\_>\\||\\|->\\|="
                limit t)
          (let ((token (match-string-no-properties 0)))
            (unless (and (equal token "|")
                         (or (memq (char-before (match-beginning 0))
                                   '(?\[ ?< ?|))
                             (memq (char-after (match-end 0))
                                   '(?\] ?> ?|))))
              (throw 'context (not (member token '("->" "=")))))))
        nil))))

(defun my/ocaml-pattern-or-binding-context-p ()
  "Return non-nil when point is in an OCaml pattern or declaration name."
  (when (derived-mode-p 'tuareg-mode)
    (or
     (condition-case nil
         (when (treesit-language-available-p 'ocaml)
           (unless (seq-find
                    (lambda (parser)
                      (eq (treesit-parser-language parser) 'ocaml))
                    (treesit-parser-list))
             (treesit-parser-create 'ocaml))
           (let ((node (treesit-node-at
                        (if (> (point) (point-min)) (1- (point)) (point))
                        'ocaml))
                 found)
             (while (and node (not found))
               (let* ((field (treesit-node-field-name node))
                      (parent (treesit-node-parent node))
                      (parent-type (and parent (treesit-node-type parent))))
                 (setq found
                       (or (equal field "pattern")
                           (and (equal field "name")
                                (string-match-p
                                 "\\(?:_binding\\|_definition\\|_declaration\\)\\'"
                                 parent-type))))
                 (setq node parent)))
             found))
       (error nil))
     (my/ocaml-incomplete-pattern-context-p))))

(defun my/completion-preview-state ()
  "Return a token identifying the current point and buffer contents."
  (cons (point) (buffer-chars-modified-tick)))

(defun my/completion-preview-dismiss ()
  "Hide the inline preview until point or the buffer changes."
  (interactive)
  (setq my/completion-preview-dismissed (my/completion-preview-state))
  (completion-preview-active-mode -1))

(defun my/completion-preview-typed-context-p ()
  "Return non-nil when this position could carry a type expectation."
  (and (fboundp 'eglot-managed-p) (eglot-managed-p)))

(defun my/completion-preview-allowed-p ()
  "Return non-nil when completion tiers are available for previewing."
  (or (not (my/completion-preview-typed-context-p))
      (my/type-completion-probe-ready-p)))

(defun my/completion-preview-ambiguous-p ()
  "Return non-nil when several candidates remain behind the preview."
  (and (cdr (completion-preview--get 'completion-preview-suffixes)) t))

(defun my/completion-preview-hole-bounds ()
  "Return the bounds of the bare hole replaced by the current preview."
  (when completion-preview--overlay
    (let* ((props (completion-preview--get 'completion-preview-props))
           (beginning (or (plist-get props :my/completion-beginning)
                          (completion-preview--get 'completion-preview-beg)))
           (end (completion-preview--get 'completion-preview-end)))
      (when (and beginning end
                 (equal (buffer-substring-no-properties beginning end) "_"))
        (cons beginning end)))))

(defun my/completion-preview-candidate (name)
  "Return the original candidate represented by preview NAME."
  (or (and my/completion-preview-candidates
           (gethash name my/completion-preview-candidates))
      name))

(defun my/completion-preview-hide-hole ()
  "Remove the visual replacement for a bare hole."
  (when (overlayp my/completion-preview-hole-overlay)
    (delete-overlay my/completion-preview-hole-overlay))
  (setq my/completion-preview-hole-overlay nil))

(defun my/completion-preview-put-rendered (text)
  "Show TEXT without losing the canonical completion preview string."
  (let* ((overlay completion-preview--overlay)
         (current (overlay-get overlay 'after-string)))
    (overlay-put overlay 'my/completion-preview-canonical current)
    (setq text (copy-sequence text))
    (unless (get-text-property 0 'face text)
      (add-text-properties 0 (length text) '(face completion-preview) text))
    (add-text-properties 0 (length text)
                         (list 'mouse-face 'completion-preview-highlight
                               'keymap my/completion-preview-mouse-map)
                         text)
    (overlay-put overlay 'my/completion-preview-rendered text)
    (overlay-put overlay 'after-string text)))

(defun my/completion-preview-reset-rendering ()
  "Forget rendering state installed by the previous preview update."
  (my/completion-preview-hide-hole)
  (when completion-preview--overlay
    (overlay-put completion-preview--overlay
                 'my/completion-preview-canonical nil)
    (overlay-put completion-preview--overlay
                 'my/completion-preview-rendered nil)))

(defun my/completion-preview-put-hole-rendered (text bounds)
  "Show TEXT in place of the bare hole at BOUNDS."
  (let ((current (overlay-get completion-preview--overlay 'after-string)))
    (overlay-put completion-preview--overlay
                 'my/completion-preview-canonical current)
    (overlay-put completion-preview--overlay 'after-string ""))
  (setq text (propertize (copy-sequence text)
                         'face 'completion-preview
                         'mouse-face 'completion-preview-highlight
                         'keymap my/completion-preview-mouse-map))
  (setq my/completion-preview-hole-overlay
        (make-overlay (car bounds) (cdr bounds)))
  (overlay-put my/completion-preview-hole-overlay 'display text))

(defun my/completion-preview-render-hole ()
  "Render the current preview as a replacement for a bare hole."
  (condition-case nil
      (when-let* ((bounds (my/completion-preview-hole-bounds))
                  (base (or (completion-preview--get
                             'completion-preview-base)
                            ""))
                  (common (substring-no-properties
                           (or (completion-preview--get
                                'completion-preview-common)
                               ""))))
        (if (my/completion-preview-ambiguous-p)
            (let ((replacement (string-remove-prefix "_" common)))
              (if (string-empty-p replacement)
                  (completion-preview-active-mode -1)
                (my/completion-preview-put-hole-rendered replacement bounds)))
          (when-let* ((suffix (car (completion-preview--get
                                    'completion-preview-suffixes)))
                      (name (concat base common suffix))
                      (candidate (my/completion-preview-candidate name))
                      (replacement
                       (or (my/type-completion-preview-text
                            candidate (car bounds))
                           (substring-no-properties candidate))))
            (my/completion-preview-put-hole-rendered replacement bounds)))
        t)
    (error nil)))

(defun my/completion-preview-trim-to-common ()
  "Reduce an ambiguous preview to the prefix every candidate shares."
  (condition-case nil
      (let* ((overlay completion-preview--overlay)
             (beg (completion-preview--get 'completion-preview-beg))
             (typed (- (overlay-start overlay) beg))
             (common (substring-no-properties
                      (or (completion-preview--get
                          'completion-preview-common)
                          ""))))
        (if (> (length common) typed)
            (my/completion-preview-put-rendered
             (propertize (substring common typed)
                         'face 'completion-preview-common))
          (completion-preview-active-mode -1)))
    (error (completion-preview-active-mode -1))))

(defun my/completion-preview-render-snippet ()
  "Show the whole expansion when exactly one candidate is being previewed."
  (condition-case nil
      (when-let* (((not (my/completion-preview-ambiguous-p)))
                  (overlay completion-preview--overlay)
                  (beg (completion-preview--get 'completion-preview-beg))
                  (beginning
                   (or (plist-get
                        (completion-preview--get 'completion-preview-props)
                        :my/completion-beginning)
                       beg))
                  (base (or (completion-preview--get
                             'completion-preview-base)
                            ""))
                  (common (substring-no-properties
                           (or (completion-preview--get
                                'completion-preview-common)
                               "")))
                  (suffix (car (completion-preview--get
                                'completion-preview-suffixes)))
                  (name (concat base common suffix))
                  (candidate (my/completion-preview-candidate name))
                  (full (my/type-completion-preview-text candidate beginning))
                  (replacement-beginning
                   (my/type-completion-replacement-beginning
                    beginning full))
                  (typed
                   (buffer-substring-no-properties
                    replacement-beginning (overlay-start overlay))))
        (if (string-prefix-p typed full)
            (my/completion-preview-put-rendered
             (substring full (length typed)))
          (my/completion-preview-put-hole-rendered
           full (cons replacement-beginning (overlay-start overlay)))))
    (error nil)))

(defun my/completion-preview-accept-hole (bounds)
  "Replace the bare hole at BOUNDS with the current preview."
  (let* ((ambiguous (my/completion-preview-ambiguous-p))
         (base (or (completion-preview--get 'completion-preview-base) ""))
         (common (substring-no-properties
                  (or (completion-preview--get 'completion-preview-common) "")))
         (suffix (car (completion-preview--get 'completion-preview-suffixes)))
         (name (and (not ambiguous) (concat base common suffix)))
         (candidate (and name (my/completion-preview-candidate name)))
         (replacement
          (if ambiguous
              (string-remove-prefix "_" common)
            (substring-no-properties candidate)))
         (exit-function
          (plist-get (completion-preview--get 'completion-preview-props)
                     :exit-function)))
    (completion-preview-active-mode -1)
    (delete-region (car bounds) (cdr bounds))
    (goto-char (car bounds))
    (insert replacement)
    (when (and candidate (functionp exit-function))
      (let ((my/completion-preview-accepting t))
        (funcall exit-function candidate 'finished)))))

(defun my/completion-preview-accept ()
  "Accept a sole candidate, or insert only an ambiguous common prefix."
  (interactive)
  (let ((hole (my/completion-preview-hole-bounds)))
    (cond
     (hole
      (my/completion-preview-accept-hole hole))
     ((my/completion-preview-ambiguous-p)
      (completion-preview-complete))
     (t
      (when-let* ((canonical
                   (completion-preview--get 'my/completion-preview-canonical)))
        (overlay-put completion-preview--overlay 'after-string canonical))
      (let ((my/completion-preview-accepting t))
        (completion-preview-insert))))))

(defun my/completion-preview-without-probe (original &rest arguments)
  "Run ORIGINAL preview update without a synchronous type probe."
  (let ((my/type-completion-inhibit-probe t))
    (if (or (equal my/completion-preview-dismissed
                   (my/completion-preview-state))
            (my/ocaml-pattern-or-binding-context-p)
            (my/type-completion-settled-p)
            (not (my/completion-preview-allowed-p)))
        (completion-preview-active-mode -1)
      (apply original arguments)
      (when completion-preview-active-mode
        (my/completion-preview-reset-rendering)
        (or (my/completion-preview-render-hole)
            (my/completion-preview-render-snippet)
            (when (my/completion-preview-ambiguous-p)
              (my/completion-preview-trim-to-common)))))))

Semantic and mode-specific completion precedes Hippie Expand and file fallback candidates in one ranked list.
#+begin_src emacs-lisp

(use-package hippie-exp
  :ensure nil
  :bind ([remap dabbrev-expand] . hippie-expand))

(use-package cape
  :after corfu
  :config
  (defvar-local my/completion-base-capfs nil)
  (defvar-local my/completion-super-capf nil)
  (defvar-local my/hippie-expand-capf-cache nil)
  (defvar my/hippie-expand-candidate-limit 256)

  (defun my/hippie-expand-buffer-eligible-p (buffer)
    "Return non-nil when Hippie Expand may search BUFFER."
    (with-current-buffer buffer
      (if hippie-expand-only-buffers
          (he-buffer-member hippie-expand-only-buffers)
        (not (he-buffer-member hippie-expand-ignore-buffers)))))

  (defun my/hippie-expand-candidates (beginning end pattern)
    "Return Hippie Expand candidates matching PATTERN between BEGINNING and END."
    (let* ((regexp (concat "\\_<" (regexp-quote pattern)
                           "\\(?:\\sw\\|\\s_\\)"
                           (if (string-empty-p pattern) "+" "*")))
           (origin (current-buffer))
           (seen (make-hash-table :test #'equal))
           (count 0)
           result)
      (cl-labels
          ((add (candidate)
             (when (and (stringp candidate)
                        (not (string-empty-p candidate))
                        (not (equal candidate pattern))
                        (< count my/hippie-expand-candidate-limit)
                        (not (gethash candidate seen)))
               (puthash candidate t seen)
               (setq count (1+ count))
               (push candidate result)))
           (valid-start-p (position)
             (or (= position (point-min))
                 (not (memq (char-syntax (char-before position)) '(?w ?_)))))
           (scan-region (buffer start stop reverse)
             (when (< count my/hippie-expand-candidate-limit)
               (with-current-buffer buffer
                 (save-excursion
                   (save-restriction
                     (widen)
                     (goto-char start)
                     (while (and (< count my/hippie-expand-candidate-limit)
                                 (if reverse
                                     (re-search-backward regexp stop t)
                                   (re-search-forward regexp stop t)))
                       (when (valid-start-p (match-beginning 0))
                         (add (match-string-no-properties 0)))))))))
           (scan-buffer (buffer &optional around)
             (if around
                 (progn
                   (scan-region buffer beginning nil t)
                   (scan-region buffer end nil nil))
               (with-current-buffer buffer
                 (scan-region buffer (point-min) nil nil))))
           (scan-kill-ring ()
             (dolist (text kill-ring)
               (let ((position 0))
                 (while (and (< count my/hippie-expand-candidate-limit)
                             (string-match regexp text position))
                   (let ((start (match-beginning 0)))
                     (when (or (zerop start)
                               (not (memq (char-syntax (aref text (1- start)))
                                          '(?w ?_))))
                       (add (match-string-no-properties 0 text))))
                   (setq position (max (1+ (match-beginning 0))
                                       (match-end 0)))))))
           (scan-abbrevs ()
             (dolist (table-symbol
                      (append '(local-abbrev-table global-abbrev-table)
                              abbrev-table-name-list))
               (when (and (boundp table-symbol)
                          (abbrev-table-p (symbol-value table-symbol)))
                 (mapatoms
                  (lambda (symbol)
                    (let ((name (symbol-name symbol)))
                      (when (or (string-empty-p pattern)
                                (string-prefix-p pattern name
                                                 completion-ignore-case))
                        (add (abbrev-expansion
                              name (symbol-value table-symbol))))))
                  (symbol-value table-symbol))))))
        (dolist (try hippie-expand-try-functions-list)
          (when (< count my/hippie-expand-candidate-limit)
            (pcase try
              ('try-expand-all-abbrevs (scan-abbrevs))
              ('try-expand-dabbrev (scan-buffer origin t))
              ('try-expand-dabbrev-visible
               (dolist (window (window-list nil 'nomini 'all-frames))
                 (scan-region (window-buffer window)
                              (window-start window) (window-end window) nil)))
              ('try-expand-dabbrev-all-buffers
               (let ((searched 0))
                 (dolist (buffer (buffer-list))
                   (when (and (not (eq buffer origin))
                              (or (not hippie-expand-max-buffers)
                                  (< searched hippie-expand-max-buffers))
                              (my/hippie-expand-buffer-eligible-p buffer))
                     (scan-buffer buffer)
                     (setq searched (1+ searched))))))
              ('try-expand-dabbrev-from-kill (scan-kill-ring)))))
        (nreverse result))))

  (defun my/type-merge-candidates (base semantic)
    "Merge BASE completion candidates with prebuilt SEMANTIC candidates."
    (let ((semantic-index (make-hash-table :test #'equal))
          (seen (make-hash-table :test #'equal))
          result)
      (dolist (candidate semantic)
        (puthash (substring-no-properties candidate) candidate semantic-index))
      (dolist (candidate base)
        (let* ((name (substring-no-properties candidate))
               (semantic-candidate (gethash name semantic-index))
               (entry (and semantic-candidate
                           (my/type-completion-entry semantic-candidate)))
               (copy (copy-sequence candidate)))
          (when (and entry (> (length copy) 0))
            (put-text-property 0 1 'my/type-search-entry entry copy))
          (puthash name t seen)
          (push copy result)))
      (dolist (candidate semantic)
        (unless (gethash (substring-no-properties candidate) seen)
          (push candidate result)))
      (nreverse result)))

    "Index the semantic results and precompute their applications."
    (let ((index (make-hash-table :test #'equal))
          (trigger-index (make-hash-table :test #'equal))
          (wanted my/type-completion-wanted)

               `((construct . ,my/type-completion-construct-results)

              (unless previous

            my/type-completion-trigger-index trigger-index
            my/type-completion-candidates
            (mapcar (lambda (name)
                      (my/type-semantic-candidate (gethash name index)))
                    (nreverse candidates)))))

  (defun my/type-completion-table (table qualifier &optional hole)
    "Augment completion TABLE with semantic candidates and type ranking.
QUALIFIER is the module path before the CAPF bounds.  HOLE non-nil means the
completion replaces a bare `_', whose text must not filter the candidates."
    (lambda (string predicate action)
      (cond
       ((eq action 'metadata)
        (let* ((metadata (completion-metadata string table predicate))
               (properties (copy-tree (cdr metadata)))
               (annotation
                (alist-get 'annotation-function properties)))
          (setf (alist-get 'category properties) 'my/type-completion)
          (setf (alist-get 'display-sort-function properties)
                #'my/completion-sort-candidates)
          (setf (alist-get 'annotation-function properties)
                (apply-partially
                 #'my/type-completion-annotation annotation))
          (cons 'metadata properties)))
       ((eq action t)
        (setq my/type-completion-qualifier (or qualifier "")
              my/type-completion-input
              (concat my/type-completion-qualifier
                      (my/type-hole-pattern string)))
        (let* ((effective (my/type-hole-pattern string))
               (completion-regexp-list
                (unless hole completion-regexp-list))
               (merged
                (my/type-merge-candidates
                 (unless (eq my/type-completion-probed 'pending)
                   (all-completions effective table predicate))
                 (all-completions
                  effective
                  (append
                   (my/type-completion-semantic-candidates qualifier)
                   (my/type-completion-literal-prefix-candidates effective))
                  predicate))))
          (when qualifier
            (setq merged
                  (seq-filter #'my/type-completion-orderless-match-p merged)))
          (when (and my/completion-hide-qualified (not qualifier))
            (setq merged (seq-remove #'my/completion-qualified-p merged)))
          merged))
       ((eq action 'lambda)
        (or (test-completion string table predicate)
            (and my/type-completion-index
                 (gethash string my/type-completion-index)
                 t)))
       (t (complete-with-action action table string predicate)))))

  (defun my/completion-ranked-capf ()
    "Return composed CAPF ranked by the language server's type ordering."
    (let* ((literal (and (derived-mode-p 'tuareg-mode)
                         (my/ocaml-literal-prefix-bounds)))
           (result (and my/completion-super-capf
                        (funcall my/completion-super-capf))))
      (when (and literal
                 (or (null result) (/= (nth 0 result) (car literal))))
        (setq result (list (car literal) (cdr literal) nil :exclusive 'no)))
      (when result
        (when-let* (((my/type-completion-ocaml-p))

* Languages

Completion behaviour shared by every language server, built on standard LSP only:

- *Server-side ranking* — candidates keep the server's own order
  (=sortText=), and are never re-scored locally. See [[Cape]].
- *Expected type at point* — =textDocument/signatureHelp= then
  =textDocument/hover=, both asynchronous, cached per position. See [[Corfu]].
- *Return-type matching* — a paren-aware split of the type on top-level =->= or
  =>=, so a candidate ranks by what it returns once applied.
- *Inline preview restraint* — the grey preview considers prefix matches only,
  chooses the best inhabited tier, and is dismissed with =S-SPC=.
- *Deprecated last* — LSP =deprecated= and tag =1= sink within their group.
- *Ordering* — type-correct candidates first, snippets next, then Hippie Expand,
  files, and ill-typed candidates; without an expected type, snippets come
  first.
- *Module-qualified toggle* — =my/completion-toggle-qualified= hides dotted
  names until a module prefix is typed.

*** Structured snippets
OCaml constructs expand into a model that TAB extends, S-TAB cycles between valid alternative forms, RET walks fields and S-RET ends. A negative prefix (C-- TAB) steps back. Each structural edit re-renders the construct and reindents it.
Yasnippet revival keeps edits individually undoable after the construct ends.

#+begin_src emacs-lisp

(defvar-local my/ocaml-snippet-sessions nil)

(defun my/ocaml-session-table ()
  "Return the structured snippet session table for the current buffer."
  (or my/ocaml-snippet-sessions
      (setq my/ocaml-snippet-sessions
            (make-hash-table :test #'eq :weakness 'key))))

(defun my/ocaml-path-get (root path)
  (let ((node root))
    (dolist (step path node)
      (setq node (if (integerp step) (nth step node) (plist-get node step))))))

(defun my/ocaml-path-set (root path value)
  (let* ((parent (my/ocaml-path-get root (butlast path)))
         (step (car (last path))))
    (if (integerp step)
        (setf (nth step parent) value)
      (plist-put parent step value))))

(defvar my/ocaml-render-counter nil)

  (let ((number (setq my/ocaml-render-counter (1+ my/ocaml-render-counter))))
    (push (cons number path) my/ocaml-render-fields)
    (format "${%d:%s}" number (or default "_"))))

(defvar my/ocaml-renderers (make-hash-table :test #'eq))

  (declare (indent 2))

  (funcall (gethash (plist-get node :kind) my/ocaml-renderers) node path))

  (let ((my/ocaml-render-counter 0)
        (my/ocaml-render-fields nil))
    (let ((text (my/ocaml-render node nil)))
      (list :template (concat text "$0")
            :fields (nreverse my/ocaml-render-fields)))))

(defun my/ocaml-hole (path value)
  (if (and value (not (equal value "_")))
      value
    (my/ocaml-field path "_")))

(my/ocaml-define-renderer type-decl (node path)
  (let ((params (plist-get node :params))
        (body (plist-get node :body)))
    (concat "type "
            (cond
             ((null params) "")
             ((cdr params)
              (concat "("
                      (string-join
                       (cl-loop for p in params for i from 0
                                collect (my/ocaml-field
                                         (append path (list :params i)) p))
                       ", ")
                      ") "))
             (t (concat (my/ocaml-field (append path (list :params 0))
                                        (car params))
                        " ")))

           "")
          (if (plist-get node :annotation)

            "")
          " = "

(my/ocaml-define-renderer param (node path)
  (let* ((name (my/ocaml-field (append path '(:name)) (plist-get node :name)))
         (annotation (plist-get node :annotation))
         (default (plist-get node :default))
         (typed (and annotation
                     (concat " : " (my/ocaml-field
                                    (append path '(:annotation)) annotation))))
         (valued (and default
                      (concat " = " (my/ocaml-field
                                     (append path '(:default)) default)))))
    (pcase (plist-get node :label)
      ('optional
       (if (or typed valued)
           (concat "?(" name typed valued ")")
         (concat "?" name)))
      ('labelled
       (if typed (concat "~(" name typed ")") (concat "~" name)))
      (_
       (if typed (concat "(" name typed ")") name)))))

(defun my/ocaml-node-from-constructible (text)
  "Return a structured node for merlin construction TEXT, when recognised."
  (when (stringp text)
    (let ((trimmed (string-trim text)))
      (cond
       ((string-match "\\`{\\(.*\\)}\\'" trimmed)
        (let ((names (delq nil
                           (mapcar (lambda (part)
                                     (let ((field (string-trim part)))
                                       (when (string-match
                                              "\\`\\([A-Za-z_][A-Za-z0-9_']*\\)[[:space:]]*="
                                              field)
                                         (match-string 1 field))))
                                   (split-string (match-string 1 trimmed)
                                                 ";" t)))))
          (when names (my/ocaml-record-literal names))))
       ((string-match "\\`\\[|\\(.*\\)|\\]\\'" trimmed)
        (my/ocaml-collection 'array))
       ((string-match "\\`\\[\\(.*\\)\\]\\'" trimmed)
        (my/ocaml-collection 'list))
       ((string-match "\\`\\([A-Z][A-Za-z0-9_'.]*\\)\\(\\([[:space:]]+_\\)*\\)\\'"
                      trimmed)
        (list :kind 'constructor-app
              :name (match-string 1 trimmed)
              :arguments (make-list
                          (length (split-string (or (match-string 2 trimmed) "")
                                                "[[:space:]]+" t))
                          "_")))))))

(defun my/ocaml-signature-items (text)
  "Return the declarations named in signature TEXT."
  (when (stringp text)
    (let* ((body (if (string-match
                      "\\_<sig\\_>\\(\\(?:.\\|\n\\)*\\)\\_<end\\_>" text)
                     (match-string 1 text)
                   text))
           (position 0)
           (found nil))
      (while (string-match
              "\\_<\\(val\\|type\\)\\_>[[:space:]]+\\([A-Za-z_][A-Za-z0-9_']*\\)"
              body position)
        (push (list (match-beginning 0) (match-string 1 body)
                    (match-string 2 body) (match-end 0))
              found)
        (setq position (match-end 0)))
      (setq found (nreverse found))
      (cl-loop for entry in found
               for index from 0
               for stop = (if (nth (1+ index) found)
                              (nth 0 (nth (1+ index) found))
                            (length body))
               collect
               (list :sort (intern (nth 1 entry))
                     :name (nth 2 entry)
                     :type (let ((rest (string-trim
                                        (substring body (nth 3 entry) stop))))
                             (when (string-prefix-p ":" rest)
                               (string-trim (substring rest 1)))))))))

(defun my/ocaml-expand (root path)
  "Create a new slot at PATH in ROOT. Return the path to focus, or nil."
  (let* ((owner-path (my/ocaml-owner-path root path))
         (slot (cdr (my/ocaml-owner root path)))
         (owner (my/ocaml-path-get root owner-path))
         (kind (and (listp owner) (plist-get owner :kind))))
    (cond
     ((and (eq kind 'type-decl) (eq slot :body)
           (let ((body (plist-get owner :body)))
             (or (null body) (equal body "_"))))
      (plist-put owner :body
                 (list :kind 'variant
                       :constructors (list (my/ocaml-constructor "Ctr1"))))
      (append owner-path '(:body :constructors 0 :name)))

     ((and (eq kind 'constructor) (eq slot :name)
           (not (plist-get owner :payload)))
      (plist-put owner :payload "t")
      (append owner-path '(:payload)))

     ((and (eq kind 'let-binding) (eq slot :name))
      (let ((params (plist-get owner :params)))
        (plist-put owner :params
                   (append params
                           (list (my/ocaml-param
                                  (my/ocaml-param-name (length params))))))
        (append owner-path (list :params (length params) :name))))

     ((and (eq kind 'let-binding) (eq slot :annotation)
           (equal (plist-get owner :annotation) "t"))
      (plist-put owner :annotation nil)
      (append owner-path '(:value)))

     ((and (eq kind 'let-binding) (eq slot :value))
      (my/ocaml-append-binding root owner-path))

     ((and (eq kind 'param) (eq slot :name)
           (eq (plist-get owner :label) 'optional)
           (not (plist-get owner :default)))
      (plist-put owner :default "0")
      (append owner-path '(:default)))

     ((and (eq kind 'param) (eq slot :name)
           (null (plist-get owner :label)))
      (let* ((binding-path (butlast owner-path 2))
             (binding (my/ocaml-path-get root binding-path)))
        (when (and (listp binding)
                   (eq (plist-get binding :kind) 'let-binding))
          (let ((params (plist-get binding :params)))
            (plist-put binding :params
                       (append params
                               (list (my/ocaml-param
                                      (my/ocaml-param-name
                                       (length params))))))
            (append binding-path
                    (list :params (length params) :name))))))

     ((and (eq kind 'param) (memq slot '(:name :default))
           (not (plist-get owner :annotation)))
      (plist-put owner :annotation "t")
      (append owner-path '(:annotation)))

     ((and (eq kind 'argument) (eq slot :value))
      (let* ((application-path (butlast owner-path 2))
             (application (my/ocaml-path-get root application-path))
             (index (car (last owner-path)))
             (arguments (and (eq (plist-get application :kind) 'application)
                             (plist-get application :arguments))))
        (when (and arguments (= index (1- (length arguments))))
          (plist-put application :arguments
                     (append arguments
                             (list (list :kind 'argument :label nil
                                         :label-kind nil :value "_"
                                         :alternatives nil))))
          (append application-path
                  (list :arguments (length arguments) :value)))))

     ((and (eq kind 'module-decl) (eq slot :name)
           (not (plist-get owner :signature)))
      (plist-put owner :signature (my/ocaml-signature-ref "S"))
      (append owner-path '(:signature :name)))

     ((and (eq kind 'module-decl) (eq slot :name)
           (eq (plist-get owner :form) 'functor))
      (let ((params (plist-get owner :params)))
        (plist-put owner :params
                   (append params
                           (list (list :kind 'module-param
                                       :name (my/ocaml-module-param-name
                                              (length params))
                                       :signature "S"))))
        (append owner-path (list :params (length params) :name))))

     ((and (eq kind 'module-param) (eq slot :signature))
      (let* ((decl-path (butlast (butlast owner-path)))
             (decl (my/ocaml-path-get root decl-path))
             (params (plist-get decl :params)))
        (plist-put decl :params
                   (append params
                           (list (list :kind 'module-param
                                       :name (my/ocaml-module-param-name
                                              (length params))
                                       :signature "S"))))
        (append decl-path (list :params (length params) :name))))

     ((and (eq kind 'module-decl) (eq slot :body)
           (let ((body (plist-get owner :body)))
             (or (null body) (equal body "_"))))
      (plist-put owner :body (my/ocaml-struct-body))
      (append owner-path '(:body :items 0 :name)))

     ((and (eq kind 'signature-ref) (memq slot '(:name)))
      (let ((constraints (plist-get owner :constraints)))
        (plist-put owner :constraints
                   (append constraints
                           (list (my/ocaml-constraint
                                  (my/ocaml-constraint-name (length constraints))))))
        (append owner-path
                (list :constraints (length constraints) :name))))

     ((and (eq kind 'match-arm) (eq slot :body))
      (let* ((expr-path (butlast (butlast owner-path)))
             (expr (my/ocaml-path-get root expr-path))
             (arms (plist-get expr :arms)))
        (plist-put expr :arms
                   (append arms
                           (list (my/ocaml-match-arm
                                  (my/ocaml-pattern-name (length arms))))))
        (append expr-path (list :arms (length arms) :pattern))))

     ((and (eq kind 'constraint) (eq slot :value))
      (let* ((ref-path (butlast (butlast owner-path)))
             (ref (my/ocaml-path-get root ref-path))
             (constraints (plist-get ref :constraints)))
        (plist-put ref :constraints
                   (append constraints
                           (list (my/ocaml-constraint
                                  (my/ocaml-constraint-name (length constraints))))))
        (append ref-path (list :constraints (length constraints) :name))))

     ((and (eq kind 'collection) (eq slot :elements))
      (let ((elements (plist-get owner :elements)))
        (plist-put owner :elements (append elements (list "_")))
        (append owner-path (list :elements (length elements)))))

     ((and (eq kind 'record-literal) (eq slot :fields))
      (my/ocaml-record-add-field owner owner-path))

     ((and (eq kind 'record-entry) (eq slot :value))
      (let* ((record-path (butlast owner-path 2))
             (record (my/ocaml-path-get root record-path))
             (index (car (last owner-path)))
             (fields (plist-get record :fields)))
        (when (= index (1- (length fields)))
          (my/ocaml-record-add-field record record-path))))

     ((and (eq kind 'constructor-app) (eq slot :arguments))
      (let ((arguments (plist-get owner :arguments)))
        (plist-put owner :arguments (append arguments (list "_")))
        (append owner-path (list :arguments (length arguments)))))

     ((my/ocaml-struct-item-path root path)
      (let* ((body-path (my/ocaml-struct-item-path root path))
             (body (my/ocaml-path-get root body-path))
             (items (plist-get body :items))
             (fresh (car (my/ocaml-body-options root path))))
        (plist-put body :items (append items (list fresh)))
        (append body-path (list :items (length items) :name)))))))

(defun my/ocaml-struct-item-path (root path)
  "Return the path of the struct body containing PATH, if any."
  (let ((remaining path)
        found)
    (while (and remaining (not found))
      (let* ((candidate (butlast remaining))
             (node (my/ocaml-path-get root candidate)))
        (when (and (listp node) (eq (plist-get node :kind) 'struct-body))
          (setq found candidate)))
      (setq remaining (butlast remaining)))
    found))

(defun my/ocaml-append-binding (root owner-path)
  "Append an `and' binding after the binding at OWNER-PATH."
  (let* ((index (car (last owner-path)))
         (group-path (butlast (butlast owner-path)))
         (group (my/ocaml-path-get root group-path))
         (bindings (plist-get group :bindings)))
    (plist-put group :bindings
               (append bindings
                       (list (my/ocaml-let-binding
                              (my/ocaml-binding-name (length bindings))))))
    (ignore index)
    (append group-path (list :bindings (length bindings) :name))))

(defconst my/ocaml-repetitions
  '((:constructors . my/ocaml-constructor-name)
    (:fields . my/ocaml-field-name)))

(defun my/ocaml-append-sibling (root path)
  "Append a sibling in the innermost repetition enclosing PATH."
  (let ((remaining path))
    (catch 'appended
      (while remaining
        (let* ((index (car (last remaining)))
               (list-path (butlast remaining))
               (slot (car (last list-path)))
               (entry (assq slot my/ocaml-repetitions)))
          (when (and (integerp index) entry)
            (let* ((container-path (butlast list-path))
                   (container (my/ocaml-path-get root container-path))
                   (items (plist-get container slot))
                   (fresh (pcase slot
                            (:constructors
                             (my/ocaml-constructor
                              (funcall (cdr entry) (length items))))
                            (:fields
                             (my/ocaml-record-field
                              (funcall (cdr entry) (length items)))))))
              (plist-put container slot (append items (list fresh)))
              (throw 'appended
                     (append container-path
                             (list slot (length items) :name))))))
        (setq remaining (butlast remaining)))
      nil)))

(defun my/ocaml-options (root path)
  "Return the alternatives available at PATH in ROOT."
  (let* ((slot (car (last path)))
         (owner-path (butlast path))
         (owner (my/ocaml-path-get root owner-path))
         (kind (plist-get owner :kind)))
    (cond
     ((and (eq kind 'type-decl) (eq slot :name))
      '(nil ("'a") ("'a" "'b")))
     ((and (eq kind 'type-decl) (memq slot '(:params)))
      '(nil ("'a") ("'a" "'b")))
     ((and (eq kind 'record-field) (eq slot :name))
      '(nil t)))))

(defun my/ocaml-cycle (root path)
  "Cycle the alternative at PATH in ROOT. Return the path to focus."
  (pcase-let* ((`(,owner . ,slot) (my/ocaml-owner root path))
               (kind (and (listp owner) (plist-get owner :kind)))
               (owner-path (my/ocaml-owner-path root path)))
    (cond
     ((and (eq kind 'type-decl) (eq slot :body)
           (let ((body (plist-get owner :body)))
             (or (null body) (equal body "_"))))
      (plist-put owner :body
                 (list :kind 'record
                       :fields (list (my/ocaml-record-field "field"))))
      (append owner-path '(:body :fields 0 :name)))

     ((and (eq kind 'record-literal) (eq slot :fields))
      (plist-put owner :base "_")
      (append owner-path '(:base)))

     ((and (eq kind 'record-literal) (eq slot :base))
      (plist-put owner :base nil)
      (append owner-path
              (if (plist-get owner :fields) '(:fields 0 :value) '(:fields))))

     ((and (eq kind 'record-entry) (eq slot :value))
      (let* ((record-path (butlast owner-path 2))
             (record (my/ocaml-path-get root record-path))
             (index (car (last owner-path)))
             (options (my/ocaml-record-unused-fields record index))
             (current (plist-get owner :name))
             (position (or (cl-position current options :test #'equal) -1))
             (next (and (> (length options) 1)
                        (nth (mod (1+ position) (length options)) options))))
        (when (and next (not (equal next current)))
          (plist-put owner :name next)
          (append owner-path '(:value)))))

     ((my/ocaml-struct-item-default-p root path)
      (let* ((body-path (my/ocaml-struct-item-path root path))
             (body (my/ocaml-path-get root body-path))
             (items (plist-get body :items))
             (index (car (last (my/ocaml-struct-item-index root path))))
             (options (my/ocaml-body-options root path))
             (current (nth index items))
             (position (or (cl-position-if
                            (lambda (option)
                              (and (eq (plist-get option :kind)
                                       (plist-get current :kind))
                                   (equal (plist-get option :name)
                                          (plist-get current :name))))
                            options)
                           -1))
             (next (nth (mod (1+ position) (length options)) options)))
        (setf (nth index items) next)
        (append body-path (list :items index :name))))

     ((and (eq kind 'type-decl) (memq slot '(:name :params)))
      (let* ((options '(nil ("'a") ("'a" "'b")))
             (current (plist-get owner :params))
             (index (or (cl-position current options :test #'equal) 0))
             (next (nth (mod (1+ index) (length options)) options)))
        (plist-put owner :params (copy-sequence next))
        (if next
            (append owner-path '(:params 0))
          (append owner-path '(:name)))))

     ((and (eq kind 'record-field) (memq slot '(:name :type)))
      (plist-put owner :mutable (not (plist-get owner :mutable)))
      (append owner-path '(:name)))

     ((and (eq kind 'constructor) (memq slot '(:name :payload :result)))
      (if (plist-get owner :result)
          (progn
            (plist-put owner :result nil)
            (append owner-path
                    (if (plist-get owner :payload) '(:payload) '(:name))))
        (plist-put owner :result
                   (or (my/ocaml-type-application
                        (my/ocaml-enclosing-type-decl root path))
                       "t"))
        (append owner-path '(:result))))

     ((eq kind 'param)
      (let* ((options '(nil labelled optional))
             (index (or (cl-position (plist-get owner :label) options) 0))
             (next (nth (mod (1+ index) (length options)) options)))
        (plist-put owner :label next)
        (unless (eq next 'optional) (plist-put owner :default nil))
        (append owner-path '(:name))))

     ((and (eq kind 'let-binding) (memq slot '(:name :annotation)))
      (let* ((group-path (butlast owner-path 2))
             (group (my/ocaml-path-get root group-path))
             (recursive-owner
              (cond
               ((and (listp group) (eq (plist-get group :kind) 'let-group))
                group)
               ((plist-get owner :standalone) owner))))
        (if recursive-owner
            (let* ((annotated (and (plist-get owner :annotation) t))
                   (current (list (and (plist-get recursive-owner :recursive) t)
                                  annotated))
                   (options '((nil nil) (t nil) (nil t) (t t)))
                   (index (or (cl-position current options :test #'equal) 0))
                   (next (nth (mod (1+ index) (length options)) options)))
              (plist-put recursive-owner :recursive (car next))
              (plist-put owner :annotation (and (cadr next) "t")))
          (plist-put owner :annotation
                     (and (not (plist-get owner :annotation)) "t")))
        (append owner-path
                (if (plist-get owner :annotation) '(:annotation) '(:name)))))

     ((and (eq kind 'module-decl) (eq slot :name))
      (let* ((options '(plain functor alias application recursive))
             (index (or (cl-position (plist-get owner :form) options) 0))
             (next (nth (mod (1+ index) (length options)) options)))
        (plist-put owner :form next)
        (plist-put owner :params
                   (when (eq next 'functor)
                     (list (list :kind 'module-param
                                 :name "X" :signature "S"))))
        (pcase next
          ('alias
           (plist-put owner :body (list :kind 'module-alias :target "M")))
          ('application
           (plist-put owner :body
                      (list :kind 'functor-app :functor "F" :argument "M")))
          (_
           (let ((body (plist-get owner :body)))
             (when (and (listp body)
                        (memq (plist-get body :kind)
                              '(module-alias functor-app)))
               (plist-put owner :body nil)))))
        (when (plist-get owner :auto-signature)
          (plist-put owner :signature nil)
          (plist-put owner :auto-signature nil))
        (when (and (eq next 'recursive) (not (plist-get owner :signature)))
          (plist-put owner :signature (my/ocaml-signature-ref "S"))
          (plist-put owner :auto-signature t))
        (append owner-path '(:name))))

     ((and (eq kind 'signature-ref) (eq slot :name))
      (let* ((decl-path (butlast owner-path))
             (decl (my/ocaml-path-get root decl-path)))
        (plist-put decl :signature
                   (list :kind 'signature-body
                         :items (list (list :kind 'val-item
                                            :name "x" :type "t"))))
        (append decl-path '(:signature :items 0 :name))))

     ((and (eq kind 'signature-body) (eq slot :items))
      (let* ((decl-path (butlast owner-path))
             (decl (my/ocaml-path-get root decl-path)))
        (plist-put decl :signature (my/ocaml-signature-ref "S"))
        (append decl-path '(:signature :name))))

     ((and (eq kind 'constraint) (memq slot '(:name :value)))
      (let ((destructive (plist-get owner :destructive))
            (what (plist-get owner :what)))
        (cond
         ((and (eq what 'type) (not destructive))
          (plist-put owner :destructive t))
         ((eq what 'type)
          (plist-put owner :what 'module)
          (plist-put owner :destructive nil))
         ((not destructive) (plist-put owner :destructive t))
         (t (plist-put owner :what 'type)
            (plist-put owner :destructive nil))))
      (append owner-path '(:name)))

     ((and (eq kind 'match-arm) (memq slot '(:pattern :guard :alias
                                             :alternative :body)))
      (let* ((options '(nil guard or-pattern as-binding))
             (current (if (plist-get owner :guard) 'guard
                        (plist-get owner :pattern-form)))
             (index (or (cl-position current options) 0))
             (next (nth (mod (1+ index) (length options)) options)))
        (plist-put owner :guard (when (eq next 'guard) "g"))
        (plist-put owner :pattern-form (unless (eq next 'guard) next))
        (when (eq next 'or-pattern)
          (plist-put owner :alternative "pat2"))
        (when (eq next 'as-binding)
          (plist-put owner :alias "x"))
        (append owner-path '(:pattern))))

     ((and (eq kind 'argument) (memq slot '(:value)))
      (when-let* ((available (plist-get owner :alternatives)))
        (let* ((current (plist-get owner :label))
               (index (or (cl-position current available :test #'equal) -1))
               (next (nth (mod (1+ index) (length available)) available)))
          (plist-put owner :label next)
          (plist-put owner :label-kind (and next 'labelled))
          (append owner-path '(:value)))))

     ((and (eq kind 'struct-body) (eq slot :items))
      nil))))

  (when my/ocaml-snippet-sessions
    (let* ((overlay (bound-and-true-p yas--active-field-overlay))
           (field (and (overlayp overlay) (overlay-get overlay 'yas--field)))
           (snippets (yas-active-snippets 'all))

            (or (and field
                     (seq-find
                      (lambda (candidate)
                        (and (gethash candidate my/ocaml-snippet-sessions)

                     (seq-find
                      (lambda (candidate)

(defun my/ocaml-join-revival-to-previous-change ()
  "Make snippet revival part of the preceding undo step."
  (when (and (consp buffer-undo-list)
             (let ((entry (car buffer-undo-list)))
               (and (consp entry)
                    (eq (car entry) 'apply)
                    (eq (cadr entry) 'yas--snippet-revive)))
             (null (cadr buffer-undo-list)))
    (setq buffer-undo-list
          (cons nil (cons (car buffer-undo-list)
                          (cddr buffer-undo-list))))))

(defun my/ocaml-restructure (mutate)
  "Apply MUTATE to the active model and make one progress step."
  (when-let* ((session (my/ocaml-active-session))
              (snippet (car session))
              (state (cdr session))
              (path (my/ocaml-current-path state)))
    (my/ocaml-sync snippet state)
    (let* ((old-root (plist-get state :root))
           (root (copy-tree old-root))
           (focus (funcall mutate root path)))
      (cond
       ((null focus) nil)
       ((equal root old-root)
        (and (not (equal focus path))
             (my/ocaml-focus-field snippet state focus)))
       (t
        (let ((region (my/ocaml-snippet-region snippet)))
          (undo-boundary)
          (atomic-change-group
            (my/ocaml-commit-snippet snippet)
            (delete-region (car region) (cdr region))
            (goto-char (car region))
            (my/ocaml-place root focus))
          (undo-boundary)
          t))))))

    (my/ocaml-commit-snippet (car session))
    (my/ocaml-join-revival-to-previous-change)
    t))

(defun my/ocaml-split-destructured-case ()
  "Split the destruct action's top-level or-pattern into separate cases."
  (when (treesit-language-available-p 'ocaml)
    (when-let* ((node (treesit-node-at (point) 'ocaml))
                (case (treesit-parent-until
                       node (lambda (candidate)
                              (string= (treesit-node-type candidate)
                                       "match_case"))
                       t))
                (pattern (treesit-node-child-by-field-name case "pattern"))
                ((string= (treesit-node-type pattern) "or_pattern"))
                (patterns (treesit-node-children pattern t))
                ((cdr patterns)))
      (let* ((start (treesit-node-start case))
             (end (treesit-node-end case))
             (tail (buffer-substring-no-properties
                    (treesit-node-end pattern) end))
             (replacement
              (mapconcat
               (lambda (alternative)
                 (concat (treesit-node-text alternative t) tail))
               patterns "\n| ")))
        (delete-region start end)
        (goto-char start)
        (insert replacement)
        (goto-char start)
        t))))

(defun my/ocaml-destruct ()
  "Destructure at point into separate cases with editable leaves."
  (declare (modes tuareg-mode))
  (interactive)
  (let ((origin (copy-marker
                 (if (region-active-p) (region-beginning) (point)) t)))
    (unwind-protect
        (progn
          (call-interactively #'ocaml-eglot-destruct)
          (goto-char origin)
          (my/ocaml-split-destructured-case)
          (when-let* ((bounds (my/ocaml-enclosing-match-bounds))
                      (text (buffer-substring-no-properties
                             (car bounds) (cdr bounds)))
                      (snippet (my/ocaml-leaf-snippet text))
                      ((not (equal snippet text))))
            (unless (bound-and-true-p yas-minor-mode) (yas-minor-mode 1))
            (delete-region (car bounds) (cdr bounds))
            (goto-char (car bounds))

** OCaml
Installed dependency sources retain Eglot navigation without standalone Flymake diagnostics.

OCaml-specific behaviour on top of the shared completion described under [[Languages]]:

- *Type-directed candidates* — =ocamllsp/typeSearch= supplies values whose return
  type matches the hole, already ranked by merlin's cost; =ocamllsp/construct=
  supplies type-correct constructions. See [[Cape]].
- *Typed holes* — a bare =_= completes as an empty prefix and is replaced by the
  chosen candidate, via a dedicated completion style. See [[Corfu]].
- *Structured snippets* — constructs are a model that =TAB= extends, =S-TAB=
  cycles between valid alternative forms, =RET= walks fields and =S-RET= ends;
  a negative prefix (=C-- TAB=) steps back. See [[Structured snippets]].
- *Snippets as candidates* — literals, operators and constructs appear in the
  completion list gated by the expected type, so =[= offers a list and =*= offers
  =(_ * _)= only where those typecheck.
- *Signature-driven bodies* — a module annotated =: S= offers the items declared
  by =S= with their names fixed and arity taken from the declared type.
- *Applications* — arity comes from the expected type, labelled arguments render
  with their labels, and optional arguments stay omitted until =TAB= adds them.
- *Menu* — =C-c .= collects snippets, type queries, navigation and refactors.

** Daily Review

=liana-review= walks today's buffer with a menu that stays open, acting on the note at
point and moving to the next. =liana-promote= moves the note into its own =final= file
node, preserving its =:ID:=, =:CREATED:=, and body. =liana-route-to-task= moves it to
=tasks.org=. Archive and discard drop it from the inbox.

#+begin_src emacs-lisp

(defun liana--heading-body ()
  (save-excursion
    (org-back-to-heading t)
    (let ((end (save-excursion (org-end-of-subtree t t) (point))))
      (org-end-of-meta-data)
      (string-trim (buffer-substring-no-properties (min (point) end) end)))))

(defun liana-promote ()
  (interactive)
  (org-back-to-heading t)
  (let* ((title (org-get-heading t t t t))
         (id (or (org-id-get) (org-id-new)))
         (created (org-entry-get (point) "CREATED"))
         (tags (seq-remove (lambda (tg) (member tg '("fleeting" "literature")))
                           (org-get-tags nil t)))
         (body (liana--heading-body))
         (file (expand-file-name
                (format "%s-%s.org" (format-time-string "%Y%m%d%H%M%S")
                        (org-roam-node-slug (org-roam-node-create :title title)))
                org-roam-directory)))
    (org-back-to-heading t)
    (org-cut-subtree)
    (with-current-buffer (find-file-noselect file)
      (erase-buffer)
      (insert ":PROPERTIES:\n:ID:       " id "\n:CLASS:    final\n"
              (if (and created (not (string-empty-p created)))
                  (concat ":CREATED:  " created "\n") "")
              ":END:\n#+title: " title "\n"
              (if tags (concat "#+filetags: :" (string-join tags ":") ":\n") "")
              "\n"
              (if (string-empty-p body) "" (concat body "\n")))
      (save-buffer)
      (org-roam-db-update-file))
    (find-file file)))

(defun liana-route-to-task ()
  (interactive)
  (org-back-to-heading t)
  (org-entry-put (point) "CLASS" "task")
  (org-todo "TODO")
  (org-toggle-tag "task" 'on)
  (org-cut-subtree)
  (with-current-buffer (find-file-noselect (liana-tasks-file))
    (goto-char (point-max))
    (unless (bolp) (insert "\n"))
    (org-paste-subtree 1)
    (save-buffer)))

(defun liana-mark-processed ()
  (interactive)
  (org-back-to-heading t)
  (org-entry-put (point) "LIANA_PROCESSED" "t"))

(defun liana-discard ()
  (interactive)
  (when (yes-or-no-p "discard subtree? ")
    (org-back-to-heading t)
    (org-cut-subtree)))

(defun liana-review-next ()
  (interactive)
  (org-next-visible-heading 1))

(defun liana-review-prev ()
  (interactive)
  (org-previous-visible-heading 1))

(defun liana--review-goto-next ()
  (goto-char (point-min))
  (catch 'found
    (while (outline-next-heading)
      (when (and (member (org-entry-get (point) "CLASS") '("fleeting" "literature"))
                 (not (org-entry-get (point) "LIANA_PROCESSED")))
        (throw 'found t)))

** Atomicity

Saving a =final= note runs =liana-atomicity-lint=: a non-blocking message if it is over

** Context Capture

Every capture stamps a folded =:CONTEXT:= drawer recording where I was: buffer, file,
major mode, project, git branch, and machine. It reads the buffer capture was invoked
from, never copies the whole buffer, and adds nothing it cannot cheaply read.

#+begin_src emacs-lisp

(defun liana--git (file)
  (when-let ((root (locate-dominating-file file ".git")))
    (let* ((head-file (expand-file-name ".git/HEAD" root))
           (head (and (file-readable-p head-file)
                      (with-temp-buffer (insert-file-contents head-file)
                        (string-trim (buffer-string)))))
           (branch (and head (string-match "ref: refs/heads/\\(.*\\)" head)
                        (match-string 1 head))))
      (list :root (file-name-nondirectory (directory-file-name root)) :branch branch))))

(defun liana-capture-context ()
  (condition-case nil
      (let ((buf (or (ignore-errors (org-capture-get :original-buffer)) (current-buffer))))
        (if (not (buffer-live-p buf)) ""
          (with-current-buffer buf
            (let* ((file (buffer-file-name))
                   (git (and file (liana--git file)))
                   (fields (delq nil
                            (list (cons "BUFFER" (buffer-name))
                                  (and file (cons "FILE" (format "[[file:%s]]" file)))
                                  (cons "MAJOR_MODE" (symbol-name major-mode))
                                  (and git (plist-get git :root) (cons "PROJECT" (plist-get git :root)))
                                  (and git (plist-get git :branch) (cons "GIT_BRANCH" (plist-get git :branch)))
                                  (cons "SYSTEM" (system-name))))))
              (concat ":CONTEXT:\n"
                      (mapconcat (lambda (kv) (format "%s: %s" (car kv) (cdr kv))) fields "\n")
                      "\n:END:")))))
    (error "")))

Direnv uses the async =envrc= PR referenced by Tyler Langlois' [[https://blog.tjll.net/asynchronous-emacs-direnv-devshells/][Asynchronous Emacs Direnv Devshells]], with Eglot deferred until the buffer environment has been applied.
#+begin_src emacs-lisp

(use-package envrc
  :config
  (setq envrc-async-processing t)

  (defvar-local max/envrc-deferred-eglot nil
    "Deferred `eglot-ensure' call waiting for envrc.")

  (defun max/envrc-loading-p ()
    "Return non-nil when current buffer's envrc is still loading."
    (and (bound-and-true-p envrc-mode)
         (fboundp 'envrc--find-env-dir)
         (envrc--find-env-dir)
         (not (memq envrc--status '(on error denied)))))

  (defun max/envrc-defer-eglot-ensure (orig &rest args)
    "Defer ORIG until envrc has applied the buffer environment."
    (if (max/envrc-loading-p)
        (setq max/envrc-deferred-eglot
              (lambda () (apply orig args)))
      (apply orig args)))

  (defun max/envrc-run-deferred-eglot (buf &optional _result)
    "Run BUF's deferred Eglot startup after envrc finishes."
    (with-current-buffer buf
      (when max/envrc-deferred-eglot
        (let ((fn max/envrc-deferred-eglot))
          (setq max/envrc-deferred-eglot nil)
          (funcall fn)))))

  (with-eval-after-load 'eglot
    (advice-add 'eglot-ensure :around #'max/envrc-defer-eglot-ensure))
  (advice-add 'envrc--apply :after #'max/envrc-run-deferred-eglot)
  (envrc-global-mode))

* Terminal
** Eat

Pure-elisp terminal, with =eat-eshell-mode= upgrading eshell visual commands. Meow normal ↔ eat-emacs-mode, meow insert ↔ eat-semi-char-mode. =<escape>= in semi-char must be rebound because eat otherwise forwards it to the terminal. =C-l= is resent on window-size change because fullscreen TUIs cache cells and leave restored rows blank.

#+begin_src emacs-lisp

(use-package eat
  :hook ((eshell-load . eat-eshell-mode)
         (eshell-load . eat-eshell-visual-command-mode)
         (eat-mode . my/disable-line-numbers)
         (eshell-mode . my/disable-line-numbers))
  :config
  (defun my/eat-meow-mode-sync ()
    "Sync eat input mode with meow state."
    (when (derived-mode-p 'eat-mode)
      (if (meow-normal-mode-p)
          (eat-emacs-mode)
        (eat-semi-char-mode))))
  (add-hook 'meow-normal-mode-hook #'my/eat-meow-mode-sync)
  (add-hook 'meow-insert-mode-hook #'my/eat-meow-mode-sync)
  (add-hook 'eat-mode-hook #'my/eat-meow-mode-sync)

  (define-key eat-semi-char-mode-map (kbd "<escape>") #'meow-insert-exit)
  (define-key eat-eshell-semi-char-mode-map (kbd "<escape>") #'meow-insert-exit)

  (defun my/eat-redraw-on-resize (window)
    "Force a full repaint of the eat program in WINDOW."
    (with-selected-window window
      (when (and (derived-mode-p 'eat-mode) eat-terminal)
        (eat-term-send-string eat-terminal "\C-l"))))
  (add-hook 'window-size-change-functions
            (lambda (frame)
              (walk-windows #'my/eat-redraw-on-resize nil frame))))

** GPTel
Interactive LLM chat uses ChatGPT OAuth with Emacs tools available.
Global and project instructions are added as context before requests.
Every request tells the model where its response will land (buffer at
point, chat buffer, echo area, kill-ring, minibuffer).

#+begin_src emacs-lisp

  (use-package gptel
    :bind (("C-c g" . gptel-menu)
           ("C-c G" . gptel-toggle-auto-agents-context))
    :config
    (require 'cl-lib)
    (require 'seq)
    (require 'gptel-context)

    (setq gptel-default-mode 'org-mode
          gptel-backend (gptel-make-openai-oauth "ChatGPT")
          gptel-model 'gpt-5.4-mini)

    (defvar gptel-auto-agents-context t)

    (defun gptel-toggle-auto-agents-context ()
      "Toggle automatic AGENTS.md context for gptel."
      (interactive)
      (setq gptel-auto-agents-context
            (not gptel-auto-agents-context))
      (unless gptel-auto-agents-context
        (gptel-remove-agents-context (current-buffer)))
      (message "GPTel AGENTS.md context %s"
               (if gptel-auto-agents-context "enabled" "disabled")))

    (defun gptel-request-buffer (args)
      "Return the request buffer named by ARGS."
      (or (plist-get (cdr args) :buffer)
          (current-buffer)))

    (defun gptel-agents-files (buffer)
      "Return global and project AGENTS.md files for BUFFER."
      (let* ((global (expand-file-name "~/.codex/AGENTS.md"))
             (default-directory
              (buffer-local-value 'default-directory buffer))
             (local-dir (locate-dominating-file default-directory "AGENTS.md"))
             (local (and local-dir (expand-file-name "AGENTS.md" local-dir))))
        (delete-dups
         (mapcar #'file-truename
                 (seq-filter #'file-readable-p
                             (delq nil (list global local)))))))

    (defun gptel-remove-agents-context (buffer)
      "Remove AGENTS.md files from gptel context for BUFFER."
      (with-current-buffer buffer
        (dolist (file (gptel-agents-files buffer))
          (setq gptel-context
                (cl-remove-if
                 (lambda (item)
                   (equal item (list file)))
                 gptel-context)))))

    (defun gptel-add-agents-context (orig &rest args)
      "Add AGENTS.md files to gptel context before calling ORIG with ARGS."
      (let ((buffer (gptel-request-buffer args)))
        (when gptel-auto-agents-context
          (with-current-buffer buffer
            (dolist (file (gptel-agents-files buffer))
              (cl-pushnew (list file) gptel-context :test #'equal)))))
      (apply orig args))

    (advice-add #'gptel-request :around #'gptel-add-agents-context)

    (defvar gptel-menu-redirect nil)

    (defun gptel-note-menu-redirect (orig args)
      "Bind the menu output redirect while ORIG runs on ARGS."
      (let ((gptel-menu-redirect
             (cond ((member "e" args) "shown in the echo area")
                   ((member "k" args) "copied to the kill-ring"))))
        (funcall orig args)))

    (defun gptel-buffer-description (buffer)
      "Describe BUFFER's file, directory, and active modes."
      (with-current-buffer buffer
        (let ((minor-modes
               (seq-filter
                (lambda (mode)
                  (and (boundp mode) (symbol-value mode)))
                (mapcar #'car minor-mode-alist))))
          (format "%s; file %s; default-directory %s; major mode %s; active minor modes %s"
                  (buffer-name)
                  (or buffer-file-name "none")
                  default-directory
                  major-mode
                  (if minor-modes
                      (mapconcat #'symbol-name minor-modes ", ")
                    "none")))))

    (defun gptel-destination-buffer (info)
      "Return the response destination buffer for request INFO."
      (let ((position (plist-get info :position)))
        (or (and (markerp position)
                 (marker-buffer position))
            (plist-get info :buffer))))

    (defun gptel-destination-description (info)
      "Describe where the response for request INFO will end up."
      (or gptel-menu-redirect
          (when-let* ((callback (plist-get info :callback))
                      ((not (memq callback '(gptel--insert-response
                                             gptel-curl--stream-insert-response)))))
            "passed to an Emacs Lisp callback, not inserted into a buffer")
          (when-let* ((buffer (gptel-destination-buffer info)))
            (with-current-buffer buffer
              (concat
               (cond ((minibufferp) "inserted into the minibuffer")
                     (gptel-mode
                      (format "appended to the dedicated gptel chat buffer %s"
                              (buffer-name)))
                     (buffer-file-name
                      (format "inserted at point in buffer %s (%s, visiting %s)"
                              (buffer-name) major-mode buffer-file-name))
                     (t (format "inserted at point in buffer %s (%s)"
                                (buffer-name) major-mode)))
               (and (plist-get info :in-place)
                    ", replacing the prompt text"))))))

    (defun gptel-add-destination-context (fsm)
      "Tell the model where the response for FSM will be displayed."
      (let* ((info (gptel-fsm-info fsm))
             (request-buffer (plist-get info :buffer))
             (destination-buffer (gptel-destination-buffer info)))
        (gptel-context--wrap-in-buffer
         (string-join
          (delq nil
                (list
                 (format "Response destination: your reply will be %s."
                         (gptel-destination-description info))
                 (when request-buffer
                   (format "Request buffer: %s."
                           (gptel-buffer-description request-buffer)))
                 (when destination-buffer
                   (format "Destination buffer: %s."
                           (gptel-buffer-description destination-buffer)))))
          "\n")
         'system)))

    (add-hook 'gptel-prompt-transform-functions
              #'gptel-add-destination-context 90)

    (advice-add #'gptel--suffix-send :around #'gptel-note-menu-redirect))

  (use-package llm-tool-collection
    :ensure nil
    :after gptel
    :config
    (setq gptel-tools
          (mapcar (apply-partially #'apply #'gptel-make-tool)
                  (llm-tool-collection-get-all))))

** LSP
Auto-start eglot in every prog-mode and (La)TeX buffer. Project roots are extended so Cargo/npm/Python/Nix subdirs are detected without a =.git=. =C-h T= echoes just the LSP-reported type at point via an eglot hover request.
#+begin_src emacs-lisp
(setq project-vc-extra-root-markers
      '("Cargo.toml" "package.json" "pyproject.toml" "flake.nix"))

(defun my/eglot-type-at-point ()
  "Echo only the LSP type signature at point."
  (interactive)
  (let* ((server (eglot--current-server-or-lose))
         (params (eglot--TextDocumentPositionParams))
         (resp (jsonrpc-request server :textDocument/hover params))
         (contents (and resp (plist-get resp :contents)))
         (text (cond ((stringp contents) contents)
                     ((and contents (plist-get contents :value))
                      (plist-get contents :value))
                     ((vectorp contents)
                      (mapconcat (lambda (c) (if (stringp c) c (plist-get c :value)))
                                 contents "\n"))))
         (blocks (and text
                      (let ((pos 0) acc)
                        (while (string-match "```[a-z-]*\n\\(\\(?:.\\|\n\\)*?\\)\n```" text pos)
                          (push (string-trim (match-string 1 text)) acc)
                          (setq pos (match-end 0)))
                        (nreverse acc)))))
    (if blocks
        (message "%s" (car (last blocks)))
      (user-error "No type info at point"))))

(use-package eglot
  :hook
  (prog-mode . eglot-ensure)
  ((LaTeX-mode latex-mode tex-mode) . eglot-ensure)
  :bind (:map eglot-mode-map ("C-h T" . my/eglot-type-at-point))
  :config
  (add-to-list 'eglot-server-programs
               '((tex-mode plain-tex-mode latex-mode LaTeX-mode)
                 "digestif")))

* Liana

A plain-=.org= Zettelkasten + GTD system. The daily loop: capture a thought, it lands in
today's buffer, and =liana-review= walks that buffer to promote, route, or archive each
entry. Most features delegate to a reused package; only the =liana-*= functions that
compose packages or add real logic are bespoke.

** Store Paths

Every path derives from =org-roam-directory=, so the store can live anywhere;
=liana--ensure-store= creates it on first use.

#+begin_src emacs-lisp

(defun liana-dailies-directory ()
  (expand-file-name (or (bound-and-true-p org-roam-dailies-directory) "daily/")
                    org-roam-directory))

(defun liana-tasks-file ()
  (expand-file-name "tasks.org" org-roam-directory))

(defun liana--refile-files ()
  (when (file-directory-p org-roam-directory)
    (directory-files-recursively org-roam-directory "\\.org\\'")))

(defun liana--inbox-files ()
  (seq-filter #'file-exists-p
              (cons (liana-tasks-file)
                    (when (file-directory-p (liana-dailies-directory))
                      (directory-files (liana-dailies-directory) t "\\.org\\'")))))

(defun liana--ensure-store ()
  (make-directory (liana-dailies-directory) t)

(defun liana--visit-tasks ()

** Daily Review

=liana-review= walks today's buffer with a menu that stays open, acting on the note at
point and moving to the next. =liana-promote= makes it a =final= note in its own file
(=org-roam-extract-subtree= does the move, keeping =:ID:= and =:CREATED:=).
=liana-route-to-task= moves it to =tasks.org=. Archive and discard drop it from the
inbox.

#+begin_src emacs-lisp

(defun liana-promote ()
  (interactive)
  (org-back-to-heading t)
  (org-entry-put (point) "CLASS" "final")
  (org-toggle-tag "fleeting" 'off)
  (unless (org-id-get) (org-id-get-create))
  (org-roam-extract-subtree))

(defun liana-route-to-task ()
  (interactive)
  (org-back-to-heading t)
  (org-entry-put (point) "CLASS" "task")
  (org-todo "TODO")
  (org-toggle-tag "task" 'on)
  (org-cut-subtree)
  (with-current-buffer (find-file-noselect (liana-tasks-file))
    (goto-char (point-max))
    (unless (bolp) (insert "\n"))
    (org-paste-subtree 1)
    (save-buffer)))

(defun liana-mark-processed ()
  (interactive)
  (org-back-to-heading t)
  (org-entry-put (point) "LIANA_PROCESSED" "t"))

(defun liana-discard ()
  (interactive)
  (when (yes-or-no-p "discard subtree? ")
    (org-back-to-heading t)
    (org-cut-subtree)))

(defun liana-review-next ()
  (interactive)
  (org-next-visible-heading 1))

(defun liana-review-prev ()
  (interactive)
  (org-previous-visible-heading 1))

(defun liana--review-goto-next ()
  (goto-char (point-min))
  (catch 'found
    (while (outline-next-heading)
      (when (and (member (org-entry-get (point) "CLASS") '("fleeting" "literature"))
                 (not (org-entry-get (point) "LIANA_PROCESSED")))
        (throw 'found t)))
    nil))

(defun liana-review-promote ()
  (interactive)
  (liana-promote)
  (liana--review-goto-next))

(defun liana-review-route ()
  (interactive)
  (liana-route-to-task)
  (liana--review-goto-next))

(defun liana-review-archive ()
  (interactive)
  (liana-mark-processed)
  (liana--review-goto-next))

(defun liana-review-discard ()
  (interactive)
  (liana-discard)
  (liana--review-goto-next))

(transient-define-prefix liana-review-menu ()
  "Liana daily review."
  [["Move"
    ("n" "next" liana-review-next :transient t)
    ("N" "prev" liana-review-prev :transient t)]
   ["Act"
    ("p" "promote -> final" liana-review-promote :transient t)
    ("r" "route -> task"    liana-review-route :transient t)
    ("a" "keep (archive)"   liana-review-archive :transient t)
    ("k" "discard"          liana-review-discard :transient t)]
   ["Surface"
    ("s" "surface related" consult-org-roam-search :transient t)]]
  [("q" "quit" transient-quit-one)])

(defun liana-review ()
  (interactive)
  (delete-other-windows)
  (org-roam-dailies-goto-today)
  (liana--review-goto-next)
  (save-selected-window (liana-inbox))
  (liana-review-menu))

** Navigation

=liana-explore= opens a note with its neighbourhood (composes =org-roam-node-find= with
the roam buffer); =liana-backlinks= groups incoming links by kind; =liana-new-final=
creates a =final= node. Plain find/search/jump delegate straight to =org-roam-node-find=
and =consult-org-roam-search=.

#+begin_src emacs-lisp

(use-package consult-org-roam
  :after org-roam
  :config (consult-org-roam-mode))

(defun liana-backlinks ()
  (interactive)
  (let* ((node (or (org-roam-node-at-point) (user-error "Not on a node")))
         (id (org-roam-node-id node))
         (rows (org-roam-db-query
                [:select [source type] :from links :where (= dest $s1)] id))
         (buf (get-buffer-create "*Liana Backlinks*")))
    (with-current-buffer buf
      (let ((inhibit-read-only t))
        (erase-buffer)
        (org-mode)
        (insert (format "#+title: Backlinks: %s\n\n" (org-roam-node-title node)))
        (if (null rows)
            (insert "No backlinks.\n")
          (dolist (grp (seq-group-by #'cadr rows))
            (insert (format "* %s\n" (car grp)))
            (dolist (row (cdr grp))
              (when-let ((src (org-roam-node-from-id (car row))))
                (insert (format "- [[id:%s][%s]]\n" (car row) (org-roam-node-title src)))))
            (insert "\n")))))
    (pop-to-buffer buf)))

(defun liana-explore ()
  (interactive)
  (org-roam-node-find)

** Atomicity

Saving a =final= note runs =liana-atomicity-lint=: a non-blocking message if it is over
its word limit or has no links yet. =liana-atomicity-dismiss= silences the length warning
by writing =:IGNORE_LENGTH_LIMIT:=.

#+begin_src emacs-lisp

(defun liana--node-link-count ()
  (save-excursion
    (goto-char (point-min))
    (let ((n 0))
      (while (re-search-forward "\\[\\[\\(?:id\\|informs\\|follows\\|supports\\|contradicts\\|relates\\):" nil t)
        (setq n (1+ n)))
      n)))

(defun liana--atomicity-warnings ()
  (let (ws (lim (cdr (assoc "final" liana-word-limits))))
    (when (and lim (not (org-entry-get (point-min) "IGNORE_LENGTH_LIMIT")))
      (let ((n (ignore-errors (liana--wc-count))))
        (when (and n (> n lim)) (push (format "over word limit (%d/%d)" n lim) ws))))
    (when (zerop (liana--node-link-count)) (push "no links" ws))
    (nreverse ws)))

             (equal (org-entry-get (point-min) "CLASS") "final"))

      (cond (ws (message "Liana: final note %s" (string-join ws "; ")))
            ((called-interactively-p 'any) (message "Liana: atomic"))))))

  (interactive)
  (org-entry-put (point-min) "IGNORE_LENGTH_LIMIT" "t")
  (message "Liana: length warning dismissed"))

(add-hook 'org-mode-hook
          (lambda () (add-hook 'after-save-hook #'liana-atomicity-lint nil t)))

** Context Capture

Every capture stamps a folded =:CONTEXT:= drawer recording where I was: buffer, file,
major mode, project, git branch, and machine. It reads the buffer capture was invoked
from, never copies the whole buffer, and adds nothing it cannot cheaply read.

#+begin_src emacs-lisp

(defun liana--git (file)
  (when-let ((root (locate-dominating-file file ".git")))
    (let* ((head-file (expand-file-name ".git/HEAD" root))
           (head (and (file-readable-p head-file)
                      (with-temp-buffer (insert-file-contents head-file)
                        (string-trim (buffer-string)))))
           (branch (and head (string-match "ref: refs/heads/\\(.*\\)" head)
                        (match-string 1 head))))
      (list :root (file-name-nondirectory (directory-file-name root)) :branch branch))))

(defun liana-capture-context ()
  (condition-case nil
      (let ((buf (or (ignore-errors (org-capture-get :original-buffer)) (current-buffer))))
        (if (not (buffer-live-p buf)) ""
          (with-current-buffer buf
            (let* ((file (buffer-file-name))
                   (git (and file (liana--git file)))
                   (fields (delq nil
                            (list (cons "BUFFER" (buffer-name))
                                  (and file (cons "FILE" (format "[[file:%s]]" file)))
                                  (cons "MAJOR_MODE" (symbol-name major-mode))
                                  (and git (plist-get git :root) (cons "PROJECT" (plist-get git :root)))
                                  (and git (plist-get git :branch) (cons "GIT_BRANCH" (plist-get git :branch)))
                                  (cons "SYSTEM" (system-name))))))
              (concat ":CONTEXT:\n"
                      (mapconcat (lambda (kv) (format "%s: %s" (car kv) (cdr kv))) fields "\n")
                      "\n:END:")))))
    (error "")))

** Dashboard

=liana-dashboard= is a read-only summary that nudges, never gates: inbox counts, stale

(defvar liana-fleeting-stale-days 3)

(defun liana--days-since (ts)
  (when ts (/ (float-time (time-subtract (current-time) (org-time-string-to-time ts)))
              86400.0)))

(defun liana--count (pred)
  (length (org-ql-select (liana--inbox-files) pred)))

(defun liana--stale-fleeting ()
  (delq nil
        (org-ql-select (liana--inbox-files)
          '(and (tags "fleeting") (not (property "LIANA_PROCESSED")) (property "CREATED"))
          :action (lambda ()
                    (let ((d (liana--days-since (org-entry-get (point) "CREATED"))))
                      (when (and d (> d liana-fleeting-stale-days))
                        (format "%s (%dd)" (org-get-heading t t t t) (floor d))))))))

(defun liana--linkless-finals ()
  (delq nil
        (mapcar (lambda (n)
                  (when (and (equal "final" (cdr (assoc "CLASS" (org-roam-node-properties n))))
                             (let ((f (org-roam-node-file n)))
                               (and f (file-exists-p f)
                                    (with-temp-buffer
                                      (insert-file-contents f)
                                      (zerop (liana--node-link-count))))))
                    (org-roam-node-title n)))
                (org-roam-node-list))))

        (insert (format "- Literature: %d\n" (liana--count '(and (tags "literature") (not (property "LIANA_PROCESSED"))))))
        (insert (format "- Tasks: %d\n" (liana--count '(and (tags "task") (not (done))))))
        (insert "\n* Stale fleeting\n")
        (dolist (s (liana--stale-fleeting)) (insert (format "- %s\n" s)))
        (insert "\n* Smells (final, no links)\n")
        (dolist (s (liana--linkless-finals)) (insert (format "- %s\n" s)))))
    (pop-to-buffer buf)))

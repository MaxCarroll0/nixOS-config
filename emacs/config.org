* Terminal
** Eat

Pure-elisp terminal, with =eat-eshell-mode= upgrading eshell visual commands. Meow normal ↔ eat-emacs-mode, meow insert ↔ eat-semi-char-mode. =<escape>= in semi-char must be rebound because eat otherwise forwards it to the terminal. =C-l= is resent on window-size change because fullscreen TUIs cache cells and leave restored rows blank.

#+begin_src emacs-lisp

(use-package eat
  :hook ((eshell-load . eat-eshell-mode)
         (eshell-load . eat-eshell-visual-command-mode)
         (eat-mode . my/disable-line-numbers)
         (eshell-mode . my/disable-line-numbers))
  :config
  (defun my/eat-meow-mode-sync ()
    "Sync eat input mode with meow state."
    (when (derived-mode-p 'eat-mode)
      (if (meow-normal-mode-p)
          (eat-emacs-mode)
        (eat-semi-char-mode))))
  (add-hook 'meow-normal-mode-hook #'my/eat-meow-mode-sync)
  (add-hook 'meow-insert-mode-hook #'my/eat-meow-mode-sync)
  (add-hook 'eat-mode-hook #'my/eat-meow-mode-sync)

  (define-key eat-semi-char-mode-map (kbd "<escape>") #'meow-insert-exit)
  (define-key eat-eshell-semi-char-mode-map (kbd "<escape>") #'meow-insert-exit)

  (defun my/eat-redraw-on-resize (window)
    "Force a full repaint of the eat program in WINDOW."
    (with-selected-window window
      (when (and (derived-mode-p 'eat-mode) eat-terminal)
        (eat-term-send-string eat-terminal "\C-l"))))
  (add-hook 'window-size-change-functions
            (lambda (frame)
              (walk-windows #'my/eat-redraw-on-resize nil frame))))

#+end_src

** LSP
Auto-start eglot in every prog-mode and (La)TeX buffer. Project roots are extended so Cargo/npm/Python/Nix subdirs are detected without a =.git=. =C-h T= echoes just the LSP-reported type at point via an eglot hover request.
#+begin_src emacs-lisp
(setq project-vc-extra-root-markers
      '("Cargo.toml" "package.json" "pyproject.toml" "flake.nix"))

(defun my/eglot-type-at-point ()
  "Echo only the LSP type signature at point."
  (interactive)
  (let* ((server (eglot--current-server-or-lose))
         (params (eglot--TextDocumentPositionParams))
         (resp (jsonrpc-request server :textDocument/hover params))
         (contents (and resp (plist-get resp :contents)))
         (text (cond ((stringp contents) contents)
                     ((and contents (plist-get contents :value))
                      (plist-get contents :value))
                     ((vectorp contents)
                      (mapconcat (lambda (c) (if (stringp c) c (plist-get c :value)))
                                 contents "\n"))))
         (blocks (and text
                      (let ((pos 0) acc)
                        (while (string-match "```[a-z-]*\n\\(\\(?:.\\|\n\\)*?\\)\n```" text pos)
                          (push (string-trim (match-string 1 text)) acc)
                          (setq pos (match-end 0)))
                        (nreverse acc)))))
    (if blocks
        (message "%s" (car (last blocks)))
      (user-error "No type info at point"))))

(use-package eglot
  :hook
  (prog-mode . eglot-ensure)
  ((LaTeX-mode latex-mode tex-mode) . eglot-ensure)
  :bind (:map eglot-mode-map ("C-h T" . my/eglot-type-at-point))
  :config
  (add-to-list 'eglot-server-programs
               '((tex-mode plain-tex-mode latex-mode LaTeX-mode)
                 "digestif")))
#+end_src

  (defun max/agda2-bootstrap ()
    "Load agda2-mode from the direnv PATH's agda, then enable it."
    (direnv-update-environment)
    (unless (fboundp 'agda2-mode)
      (let ((coding-system-for-read 'utf-8)
            (loc (string-trim (shell-command-to-string "agda-mode locate"))))
        (when (file-exists-p loc)
          (load-file loc))))
    (if (fboundp 'agda2-mode)
        (agda2-mode)
      (message "No agda on PATH; agda2-mode not loaded")))

  (dolist (pat '("\\.agda\\'" "\\.lagda\\'" "\\.lagda\\.md\\'" "\\.lagda\\.tex\\'"))
    (add-to-list 'auto-mode-alist (cons pat #'max/agda2-bootstrap)))

#+end_src
